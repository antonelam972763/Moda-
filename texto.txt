        // Cart Logic
        function addToCart(productId) {
            const p = products.find(prod => prod.id === productId);
            if(!p) return;

            const cartItemKey = `${p.id}-${currentSelectedSize}-${currentSelectedColor}`;
            const existing = cart.find(item => item.key === cartItemKey);

            if(existing) {
                existing.qty += 1;
            } else {
                cart.push({
                    key: cartItemKey,
                    id: p.id,
                    name: p.name,
                    price: p.price,
                    image: p.images[0],
                    size: currentSelectedSize,
                    color: currentSelectedColor,
                    qty: 1
                });
            }

            updateCartUI();
            closeProductModal();
            showToast(`¡${p.name} (Talla ${currentSelectedSize}) agregado al carrito!`);
            toggleCartDrawer();
        }

        function updateCartUI() {
            const countEl = document.getElementById('cart-count');
            const itemsEl = document.getElementById('cart-items');
            const subtotalEl = document.getElementById('cart-subtotal');
            const totalEl = document.getElementById('cart-total');

            const totalCount = cart.reduce((sum, item) => sum + item.qty, 0);
            countEl.textContent = totalCount;

            if(cart.length === 0) {
                itemsEl.innerHTML = `
                    <div class="text-center py-16 space-y-3">
                        <div class="w-16 h-16 bg-purple-100 text-purple-500 rounded-full flex items-center justify-center mx-auto text-2xl">
                            <i class="fa-solid fa-bag-shopping"></i>
                        </div>
                        <p class="font-bold text-purple-950 font-heading">Tu carrito está vacío</p>
                        <p class="text-xs text-purple-700">Añade ropita o accesorios para consentir a tu perrito.</p>
                    </div>
                `;
                subtotalEl.textContent = '$0';
                totalEl.textContent = '$0';
                return;
            }

            let subtotal = 0;
            itemsEl.innerHTML = cart.map(item => {
                subtotal += item.price * item.qty;
                return `
                    <div class="flex items-center gap-4 bg-purple-50/60 p-4 rounded-2xl border border-purple-100">
                        <img src="${item.image}" class="w-16 h-16 object-cover rounded-xl bg-white">
                        <div class="flex-1">
                            <h4 class="font-bold text-sm text-purple-950">${item.name}</h4>
                            <p class="text-xs text-purple-700">Talla: <span class="font-semibold">${item.size}</span> | Color: <span class="font-semibold">${item.color}</span></p>
                            <p class="text-sm font-extrabold text-purple-900 mt-1">$${(item.price * item.qty).toLocaleString('es-CL')}</p>
                        </div>
                        <div class="flex flex-col items-end justify-between h-full">
                            <button onclick="removeFromCart('${item.key}')" class="text-purple-400 hover:text-red-500 transition text-xs">
                                <i class="fa-solid fa-trash"></i>
                            </button>
                            <div class="flex items-center gap-2 bg-white px-2 py-1 rounded-xl border border-purple-200 mt-2">
                                <button onclick="updateQty('${item.key}', -1)" class="text-purple-700 font-bold px-1">-</button>
                                <span class="text-xs font-bold text-purple-950">${item.qty}</span>
                                <button onclick="updateQty('${item.key}', 1)" class="text-purple-700 font-bold px-1">+</button>
                            </div>
                        </div>
                    </div>
                `;
            }).join('');

            subtotalEl.textContent = `$${subtotal.toLocaleString('es-CL')}`;
            totalEl.textContent = `$${subtotal.toLocaleString('es-CL')}`;
        }

        function updateQty(key, delta) {
            const item = cart.find(i => i.key === key);
            if(!item) return;
            item.qty += delta;
            if(item.qty <= 0) {
                cart = cart.filter(i => i.key !== key);
            }
            updateCartUI();
        }

        function removeFromCart(key) {
            cart = cart.filter(i => i.key !== key);
            updateCartUI();
        }

        function toggleCartDrawer() {
            const drawer = document.getElementById('cart-drawer');
            const panel = document.getElementById('cart-panel');
            if(drawer.classList.contains('hidden')) {
                drawer.classList.remove('hidden');
                drawer.classList.add('flex');
                setTimeout(() => panel.classList.remove('translate-x-full'), 10);
            } else {
                panel.classList.add('translate-x-full');
                setTimeout(() => {
                    drawer.classList.remove('flex');
                    drawer.classList.add('hidden');
                }, 300);
            }
        }

        // Size Guide Modal
        function openSizeGuideModal() {
            document.getElementById('size-guide-modal').classList.remove('hidden');
            document.getElementById('size-guide-modal').classList.add('flex');
        }
        function closeSizeGuideModal() {
            document.getElementById('size-guide-modal').classList.remove('flex');
            document.getElementById('size-guide-modal').classList.add('hidden');
        }

        // Checkout Modal
        function openCheckoutModal() {
            if(cart.length === 0) {
                showToast("Tu carrito está vacío. Añade productos primero.");
                return;
            }
            toggleCartDrawer();
            document.getElementById('checkout-modal').classList.remove('hidden');
            document.getElementById('checkout-modal').classList.add('flex');
        }
        function closeCheckoutModal() {
            document.getElementById('checkout-modal').classList.remove('flex');
            document.getElementById('checkout-modal').classList.add('hidden');
        }

        // Process Order and route to RUT 23.004.582-8
        function processOrder(event) {
            event.preventDefault();
            const name = document.getElementById('cust-name').value;
            const phone = document.getElementById('cust-phone').value;
            const email = document.getElementById('cust-email').value;
            const address = document.getElementById('cust-address').value;
            const bank = document.getElementById('payment-bank').value;

            const total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            const orderId = 'GS-' + Math.floor(100000 + Math.random() * 900000);

            closeCheckoutModal();

            // Populate success box with destination RUT info
            const detailsBox = document.getElementById('order-details-box');
            detailsBox.innerHTML = `
                <p><strong>N° Pedido:</strong> ${orderId}</p>
                <p><strong>Cliente:</strong> ${name}</p>
                <p><strong>Destino Fondos:</strong> RUT 23.004.582-8</p>
                <p><strong>Banco:</strong> ${bank.toUpperCase()}</p>
                <p><strong>Total Pagado:</strong> $${total.toLocaleString('es-CL')}</p>
                <p><strong>Dirección:</strong> ${address}</p>
            `;

            document.getElementById('success-modal').classList.remove('hidden');
            document.getElementById('success-modal').classList.add('flex');

            // Reset cart
            cart = [];
            updateCartUI();
        }

        function closeSuccessModal() {
            document.getElementById('success-modal').classList.remove('flex');
            document.getElementById('success-modal').classList.add('hidden');
        }

        // Toast Notification
        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').textContent = msg;
            toast.classList.remove('translate-y-24', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-24', 'opacity-0');
            }, 3500);
        }
